-- عرض جميع منتجات Pablo
SELECT pd.title, p.price, md.path
FROM products p
JOIN product_detail pd ON p.id = pd.product_id
JOIN brands b ON p.brand_id = b.id
JOIN media m ON p.media_id = m.id
JOIN media_detail md ON m.id = md.media_id
WHERE b.name = 'Pablo' AND md.media_type = 'large';

-- عرض أغلى 5 منتجات
SELECT pd.title, b.name, p.price
FROM products p
JOIN product_detail pd ON p.id = pd.product_id
JOIN brands b ON p.brand_id = b.id
ORDER BY p.price DESC
LIMIT 5;

-- إحصائيات حسب العلامة التجارية
SELECT
    b.name,
    COUNT(p.id) as products,
    AVG(p.price) as avg_price,
    MIN(p.price) as min_price,
    MAX(p.price) as max_price
FROM brands b
LEFT JOIN products p ON b.id = p.brand_id
GROUP BY b.id;
